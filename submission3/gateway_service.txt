========================================================================
databricks ai-gateway create-model-service \
  schemas/reyden_whisperers_catalog.meridian_ai_gateway meridian-rm-gw-acs \
  --json '{"config":{"routing":{"destinations":[{
      "destination_type":"DESTINATION_TYPE_PAY_PER_TOKEN_FOUNDATION_MODEL",
      "name":"system.ai.databricks-claude-opus-4-8",
      "pay_per_token_config":{"model":"models/system.ai.databricks-claude-opus-4-8"},
      "traffic_percentage":100}]}},
     "supported_api_types":["openai/v1/chat/completions"]}' \
  --profile reyden-whisperers

# Result: model-services/reyden_whisperers_catalog.meridian_ai_gateway.meridian-rm-gw-acs
# OpenAI-compatible endpoint: {host}/ai-gateway/mlflow/v1/chat/completions  (model = the full 3-part name)
