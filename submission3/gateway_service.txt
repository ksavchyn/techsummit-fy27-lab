# 2) Custom guardrail function (blocks all-data / unfiltered reads) — see below.
#    Attached to the model service as a service policy (Governance > Policies).
CREATE OR REPLACE FUNCTION reyden_whisperers_catalog.meridian_ai_gateway.guard_block_all_data(event VARIANT)
RETURNS STRUCT<result STRING, reason STRING> LANGUAGE SQL
RETURN CASE WHEN
     lower(cast(event as string)) LIKE '%all data%'
  OR lower(cast(event as string)) LIKE '%all customers%'
  OR lower(cast(event as string)) LIKE '%every customer%'
  OR lower(cast(event as string)) LIKE '%entire dataset%'
  OR lower(cast(event as string)) LIKE '%read everything%'
  OR lower(cast(event as string)) LIKE '%unfiltered%'
  OR lower(cast(event as string)) LIKE '%select *%'
  THEN NAMED_STRUCT('result','deny','reason','Blocked: attempts to read all/unfiltered Lakebase data')
  ELSE NAMED_STRUCT('result','allow','reason','') END;
